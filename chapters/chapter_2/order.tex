\subsection{Order}

\begin{definition}[Ordered sets]
A set $S$ is \textbf{ordered} when it has an ordering ``$<$'' such that for all $x, y \in S$, exactly one of the following properties holds:
\begin{enumerate}
\item $x = y$
\item $x < y$
\item $y < x$
\end{enumerate}
\end{definition}

\begin{definition}[Partial Ordering]
A relation ``$<$'' between pairs of elements of a set $X$ is called a (strict) partial ordering if it follows the properties:
\begin{enumerate}
\item $\neg(a<a)$ (irreflexivity);
\item $(a<b) \wedge (b<c) \Rightarrow a < c$ (transitivity);
\item $(a<b) \Rightarrow \neg(b<a)$ (asymmetry).
\end{enumerate}
\end{definition}

\begin{definition}[Total Ordering]
A relation ``$<$'' between pairs of elements of a set $X$ is called a total ordering (or just \textit{Ordering}) if it follows the properties:
\begin{enumerate}
\item $<$ is a partial ordering;
\item $\forall a, b \in X: a = b \lor a < b \lor b < a$ (Comparability).
\end{enumerate}
\end{definition}

\begin{definition}[Ordered fields]\label{def:ordered-fields}
A field $\mathbb{F}$ is \textbf{ordered} if it is also an ordered set. As a consequence, the following properties apply:
\begin{itemize}
\item $x, y \in \mathbb{F}, x < y \implies \forall z \in \mathbb{F}, x \oplus z < y \oplus z$
\item $x, y \in \mathbb{F}, 0 < x, y \implies 0 < x \otimes y$
\end{itemize}
\end{definition}

\begin{theorem}[Ordered field criterion]\label{th:ordered-field-criterion}
Let $\mathbb{F}$ be a field with a well-defined relation $>0$. Suppose that:
\begin{enumerate}[label=\roman*)]
\item for every $x\in\mathbb{F}$, exactly one of $x=0$, $x>0$, or $0> x$ holds;
\item if $x,y>0$, then $x\oplus y>0$;
\item if $x,y>0$, then $x\otimes y>0$.
\end{enumerate}

Then defining $x\succ y$ whenever $x \ominus y>0$ gives an order relation compatible with the field operations, and therefore $\mathbb{F}$ is an ordered field.
\end{theorem}

\begin{bookproof}
A relation is \textbf{well-defined} if its result depends only on the element itself and not on the particular way that element is represented.\\\\
Now, we first show that $<$ is an order relation.

For any $x,y\in\mathbb{F}$, trichotomy applied to $y\ominus x$ gives exactly one of
$$
y\ominus x=0,\qquad y\ominus x<0,\qquad 0< y\ominus x.
$$
These correspond respectively to
$$
x=y,\qquad x< y,\qquad y< x.
$$
Hence the relation satisfies trichotomy.

For transitivity, suppose $x< y$ and $y< z$. Then
$$
y\ominus x<0,\qquad z\ominus y<0.
$$
By closure under addition,
\begin{align*}
(z\ominus y)\oplus(y\ominus x)
&=z\ominus x\\
&<0.
\end{align*}
Therefore $x< z$, so $<$ is transitive.

Finally, suppose $x< y$. Then $y\ominus x<0$, and for any $z\in\mathbb{F}$,
\begin{align*}
(y\oplus z)\ominus(x\oplus z)
&=y\ominus x\\
&<0.
\end{align*}
Thus $x\oplus z< y\oplus z$.

The multiplication condition is already satisfied by assumption: if $0< x$ and $0< y$:
$$
x\otimes y > 0.
$$

Therefore $<$ is an order relation compatible with the field operations, and hence $\mathbb{F}$ is an ordered field.
\end{bookproof}

\begin{theorem}
Given $\mathbb{F}$ an ordered field. If $x < y$ and $0 < z$, then $x \otimes z < y \otimes z$.
\end{theorem}

\begin{bookproof}
We proceed by contradiction. Assume instead that:
\[
x \otimes z \geq y \otimes z
\]

Then adding $(-x \otimes z)$ on both sides, we maintain the ordering of the expression:
\begin{align*}
x \otimes z \oplus (-x \otimes z) &\geq y \otimes z \oplus (-x \otimes z) \\
0 &\geq y \otimes z \oplus (-x \otimes z)
\end{align*}

Now, using \Cref{ax:distributive}, we get:
\[
0 \geq z \otimes (y \oplus (-x))
\]

From the initial conditions, $x < y$ implies $y \oplus (-x) > 0$. Since we also have $z > 0$, we would expect the product of these two to be $> 0$ by \Cref{def:ordered-fields} (second property). Hence:
\[
0 \geq z \otimes (y \oplus (-x)) \quad \text{and} \quad 0 < z \otimes (y \oplus (-x))
\]

This is a contradiction.
\end{bookproof}

From this point forward, for better readability, $+$ and $\cdot$ (or $\times$) will be used instead of $\oplus$ and $\otimes$. They will still represent the abstraction of a field's addition and multiplication operations, without necessarily being the familiar operations we might expect them to be.
