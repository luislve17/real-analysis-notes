\subsection{Problems \& Exercises}

\begin{problem}
Show that if $I$ is any system of nested closed intervals, then
$$
\sup\{a \in \mathbb{R} \mid [a,b] \in I\} = \alpha \le \beta = \inf\{b \in \mathbb{R} \mid [a,b] \in I\}
$$
and
$$
[\alpha, \beta] = \bigcap_{[a,b] \in I} [a,b]
$$
\end{problem}


\begin{problem}
Let $S = \left[ 0, 2 \right] \cap \mathbb{Q}$. Construct a family of open intervals in \mathbb{R} that covers $S$ but has no finite subfamily covering $S$.
\end{problem}

\begin{problem}
Let $S$ be a bounded infinite set of real numbers, and let $L$ be the set of its limit points. Prove that $L$ has a largest element, that is, a limit point $p \in S$ with $q \leq p$ for every limit point $q$.
\end{problem}

\begin{problem}
Let a system of open intervals $\{U_\alpha \}$ cover $[a, b]$. Prove there is a number $\delta > 0$ such that every subinterval of $[a, b]$ of length less than $\delta$ lies entirely inside a single interval $\{U_\alpha \}$ of the system.
\end{problem}

\begin{problem}
Suppose the points of $[0, 1]$ could be listed as $x_1, x_2, x_3, \dots$ (every point of $[0, 1]$ appears somewhere in the list). Construct a point of $[0, 1]$ that is not in the list, a contradiction.
\end{problem}

\begin{problem}
Let $f \colon [a,b] \to \mathbb{R}$ be a function such that $f(a) < 0 < f(b)$, and such that for every $x \in [a,b]$ with $f(x) \neq 0$ there exists $\delta > 0$ satisfying
\[
\operatorname{sgn} f(y) = \operatorname{sgn} f(x) \quad \text{for all } y \in (x - \delta,\, x + \delta) \cap [a,b].
\]
Prove that there exists $c \in [a,b]$ such that $f(c) = 0$.
\end{problem}
